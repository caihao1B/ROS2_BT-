cmake_minimum_required(VERSION 3.16)
project(coverage_planner)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(Fields2Cover REQUIRED)
find_package(ortools CONFIG REQUIRED)
add_executable(planner_backend src/planner_backend.cpp)
target_link_libraries(planner_backend Fields2Cover::Fields2Cover ortools::ortools)
install(TARGETS planner_backend DESTINATION lib/${PROJECT_NAME})
ament_package()
add_executable(planner_lesson_s12 ../../../lessons/S12/lesson.cpp)
target_link_libraries(planner_lesson_s12 Fields2Cover::Fields2Cover)
