cmake_minimum_required(VERSION 3.16)
project(map_manager)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(mapping_manager src/mapping_manager.cpp)
ament_target_dependencies(mapping_manager rclcpp std_msgs std_srvs nav_msgs)
target_link_libraries(mapping_manager nlohmann_json::nlohmann_json)
install(TARGETS mapping_manager DESTINATION lib/${PROJECT_NAME})
find_package(RTABMap REQUIRED COMPONENTS core)
# Jazzy's RTABMap/PCL export passes this bare token to add_definitions().
# Normalize locally; do not modify system packages or compiler safety settings.
remove_definitions(EIGEN_MPL2_ONLY)
add_compile_definitions(EIGEN_MPL2_ONLY)
add_executable(graph_probe src/graph_probe.cpp)
target_link_libraries(graph_probe rtabmap::core nlohmann_json::nlohmann_json)
install(TARGETS graph_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
