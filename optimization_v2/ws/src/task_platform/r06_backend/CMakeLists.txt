cmake_minimum_required(VERSION 3.16)
project(r06_common_supervisor)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(behaviortree_cpp 4.10.0 EXACT REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(r06_supervisor supervisor.cpp)
ament_target_dependencies(r06_supervisor rclcpp std_msgs std_srvs behaviortree_cpp)
target_link_libraries(r06_supervisor nlohmann_json::nlohmann_json)
add_executable(r06_policy_test policy_test.cpp)
ament_target_dependencies(r06_policy_test behaviortree_cpp)
target_link_libraries(r06_policy_test nlohmann_json::nlohmann_json)
target_compile_options(r06_policy_test PRIVATE -UNDEBUG)
