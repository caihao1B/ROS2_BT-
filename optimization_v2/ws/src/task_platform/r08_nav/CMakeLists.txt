cmake_minimum_required(VERSION 3.16)
project(r08_navigation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(r08_observed_prior SHARED observed_prior.cpp)
ament_target_dependencies(r08_observed_prior nav2_costmap_2d nav_msgs pluginlib tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS r08_observed_prior LIBRARY DESTINATION lib)
ament_package()
