cmake_minimum_required(VERSION 3.16)
project(r10_connector_navigation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
add_library(r10_path_curvature SHARED controller.cpp)
ament_target_dependencies(r10_path_curvature nav2_regulated_pure_pursuit_controller nav2_core nav2_costmap_2d pluginlib)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS r10_path_curvature LIBRARY DESTINATION lib)
ament_package()
