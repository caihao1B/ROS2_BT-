cmake_minimum_required(VERSION 3.16)
project(r07_navigation)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
add_library(r07_exact_goal SHARED exact_goal.cpp)
ament_target_dependencies(r07_exact_goal nav2_smac_planner nav2_core nav2_costmap_2d pluginlib)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS r07_exact_goal LIBRARY DESTINATION lib)
ament_package()
