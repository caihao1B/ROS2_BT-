cmake_minimum_required(VERSION 3.16)
project(showcase_demo)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(Protobuf_PROTOC_EXECUTABLE /usr/bin/protoc CACHE FILEPATH "Gazebo system protoc" FORCE)
find_package(Protobuf MODULE REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(showcase_actors SHARED src/ReactiveActors.cc)
target_link_libraries(showcase_actors gz-sim8::gz-sim8 gz-plugin2::gz-plugin2 gz-transport13::gz-transport13)
install(TARGETS showcase_actors DESTINATION lib)
install(DIRECTORY scripts tests DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
ament_package()
