cmake_minimum_required(VERSION 3.16)
project(mission_bt)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
 "${CMAKE_CURRENT_SOURCE_DIR}/scripts/terrain_source14.py"
 "${CMAKE_CURRENT_SOURCE_DIR}/../terrain_perception/src/terrain_node.cpp"
 "${CMAKE_CURRENT_SOURCE_DIR}/../navigation_adapter/src/command_gate.cpp")
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(behaviortree_cpp 4.10.0 EXACT REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters segmentation search)
find_package(Eigen3 REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_smac_planner REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
remove_definitions(EIGEN_MPL2_ONLY)
add_compile_definitions(EIGEN_MPL2_ONLY)
add_executable(mission_bt src/mission_bt.cpp)
target_include_directories(mission_bt PRIVATE include)
ament_target_dependencies(mission_bt rclcpp std_msgs behaviortree_cpp)
target_link_libraries(mission_bt nlohmann_json::nlohmann_json)
# Preserve the qualified stop latch, clock, owner and feedback gates unchanged.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../navigation_adapter/src/command_gate.cpp" source)
set(old "else if(!fresh(\"tool\",.25)||!tool.value(\"stopped_confirmed\",false)||tool.value(\"tool_on_requested\",true)||tool.value(\"active\",true))reason=\"TOOL_NOT_STOPPED\";")
string(FIND "${source}" "${old}" index)
if(index EQUAL -1)
 message(FATAL_ERROR "S13 tool gate adapter mismatch")
endif()
string(REPLACE "${old}" "else if(!fresh(\"tool\",.25)||!s14::toolPermit(tool,lease))reason=\"TOOL_WORK_INTERLOCK\";" source "${source}")
# Command margin below S10's unchanged 0.12 m/s measured-speed admission.
string(REPLACE "std::min({.12,s.value" "std::min({lease.value(\"work\",false)?.11:.12,s.value" source "${source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/gate.cpp" "#include \"tool_permit.hpp\"\n${source}")
add_executable(command_gate "${CMAKE_CURRENT_BINARY_DIR}/gate.cpp")
target_include_directories(command_gate PRIVATE include ../navigation_adapter/src)
ament_target_dependencies(command_gate rclcpp std_msgs nav_msgs geometry_msgs)
target_link_libraries(command_gate nlohmann_json::nlohmann_json)
execute_process(COMMAND /usr/bin/python3 "${CMAKE_CURRENT_SOURCE_DIR}/scripts/terrain_source14.py" "${CMAKE_CURRENT_BINARY_DIR}/terrain14.cpp" COMMAND_ERROR_IS_FATAL ANY)
add_executable(terrain14 "${CMAKE_CURRENT_BINARY_DIR}/terrain14.cpp")
target_include_directories(terrain14 PRIVATE ../terrain_perception/include ${PCL_INCLUDE_DIRS})
ament_target_dependencies(terrain14 rclcpp sensor_msgs nav_msgs std_msgs pcl_conversions)
target_link_libraries(terrain14 ${PCL_LIBRARIES} Eigen3::Eigen nlohmann_json::nlohmann_json)
add_executable(terrain_plane_test tests/terrain_plane.cpp)
target_include_directories(terrain_plane_test PRIVATE include ../terrain_perception/include)
target_link_libraries(terrain_plane_test Eigen3::Eigen)
target_compile_options(terrain_plane_test PRIVATE -UNDEBUG)
install(TARGETS mission_bt command_gate terrain14 terrain_plane_test DESTINATION lib/${PROJECT_NAME})
add_library(ackermann_planner14 SHARED src/ackermann_planner14.cpp)
target_include_directories(ackermann_planner14 PRIVATE include ../navigation_adapter/src)
ament_target_dependencies(ackermann_planner14 rclcpp std_msgs nav2_core nav2_costmap_2d pluginlib)
target_link_libraries(ackermann_planner14 nlohmann_json::nlohmann_json)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS ackermann_planner14 LIBRARY DESTINATION lib)
add_executable(context_clock_test tests/context_clock.cpp)
target_include_directories(context_clock_test PRIVATE include)
target_link_libraries(context_clock_test nlohmann_json::nlohmann_json)
target_compile_options(context_clock_test PRIVATE -UNDEBUG)
install(TARGETS context_clock_test DESTINATION lib/${PROJECT_NAME})
add_executable(planner_replay14 tests/planner_replay14.cpp)
ament_target_dependencies(planner_replay14 rclcpp nav2_smac_planner nav2_costmap_2d)
target_link_libraries(planner_replay14 nlohmann_json::nlohmann_json)
install(TARGETS planner_replay14 DESTINATION lib/${PROJECT_NAME})
add_executable(path_retention_test tests/path_retention.cpp)
target_include_directories(path_retention_test PRIVATE include)
ament_target_dependencies(path_retention_test nav_msgs nav2_costmap_2d)
target_compile_options(path_retention_test PRIVATE -UNDEBUG)
install(TARGETS path_retention_test DESTINATION lib/${PROJECT_NAME})
add_executable(risk_evidence_test tests/risk_evidence.cpp)
target_include_directories(risk_evidence_test PRIVATE include ../terrain_perception/include)
target_link_libraries(risk_evidence_test Eigen3::Eigen nlohmann_json::nlohmann_json)
target_compile_options(risk_evidence_test PRIVATE -UNDEBUG)
install(TARGETS risk_evidence_test DESTINATION lib/${PROJECT_NAME})
add_executable(local_surface_test tests/local_surface.cpp)
target_include_directories(local_surface_test PRIVATE include ../terrain_perception/include)
target_link_libraries(local_surface_test Eigen3::Eigen nlohmann_json::nlohmann_json)
target_compile_options(local_surface_test PRIVATE -UNDEBUG)
install(TARGETS local_surface_test DESTINATION lib/${PROJECT_NAME})
ament_package()
