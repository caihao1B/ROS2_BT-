cmake_minimum_required(VERSION 3.16)
project(prediction_risk_guard)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(behaviortree_cpp 4.10.0 EXACT REQUIRED)
find_package(nlohmann_json REQUIRED)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS scripts/generate15.py ../mission_bt/src/mission_bt.cpp ../navigation_adapter/src/command_gate.cpp)
execute_process(COMMAND /usr/bin/python3 ${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate15.py ${CMAKE_CURRENT_BINARY_DIR} COMMAND_ERROR_IS_FATAL ANY)
add_executable(guard15 src/guard15.cpp)
add_executable(command_gate15 ${CMAKE_CURRENT_BINARY_DIR}/gate15.cpp)
add_executable(mission_bt15 ${CMAKE_CURRENT_BINARY_DIR}/bt15.cpp)
foreach(target guard15 command_gate15 mission_bt15)
 target_include_directories(${target} PRIVATE include ../mission_bt/include ../navigation_adapter/src)
 ament_target_dependencies(${target} rclcpp std_msgs nav_msgs geometry_msgs behaviortree_cpp)
 target_link_libraries(${target} nlohmann_json::nlohmann_json)
endforeach()
install(TARGETS guard15 command_gate15 mission_bt15 DESTINATION lib/${PROJECT_NAME})
ament_package()
