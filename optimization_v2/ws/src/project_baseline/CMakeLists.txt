cmake_minimum_required(VERSION 3.16)
project(project_baseline)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(behaviortree_cpp REQUIRED)
find_package(Fields2Cover REQUIRED)
find_library(METIS_LIBRARY NAMES metis REQUIRED)
add_executable(baseline_bt src/bt_engine.cpp)
target_link_libraries(baseline_bt behaviortree_cpp::behaviortree_cpp)
add_executable(dependency_probe src/dependency_probe.cpp)
target_link_libraries(dependency_probe Fields2Cover::Fields2Cover ${METIS_LIBRARY})
add_executable(isolation_publisher src/isolation_publisher.cpp)
ament_target_dependencies(isolation_publisher rclcpp std_msgs)
install(TARGETS baseline_bt dependency_probe isolation_publisher DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY fixtures DESTINATION share/${PROJECT_NAME})
ament_package()
