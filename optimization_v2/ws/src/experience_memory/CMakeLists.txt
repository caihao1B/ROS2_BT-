cmake_minimum_required(VERSION 3.16)
project(experience_memory)
find_package(ament_cmake REQUIRED)
add_library(memory_admission SHARED src/admission.cpp)
target_include_directories(memory_admission PRIVATE include)
target_compile_features(memory_admission PRIVATE cxx_std_17)
target_compile_options(memory_admission PRIVATE -Wall -Wextra -O2)
install(TARGETS memory_admission LIBRARY DESTINATION lib)
install(DIRECTORY scripts include DESTINATION share/${PROJECT_NAME})
ament_package()
