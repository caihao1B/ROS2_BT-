# Reuse S09 risk/stop logic; adapt its self mask and add fresh confirmation metadata.
# Fail closed if the upstream source no longer matches the reviewed patch.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../terrain_perception/src/terrain_node.cpp" terrain_source)
set(old_mask "if(v.x()>-.65&&v.x()<1.55&&std::abs(v.y())<.7&&v.z()<1.1)continue;")
string(FIND "${terrain_source}" "${old_mask}" mask_index)
if(mask_index EQUAL -1)
  message(FATAL_ERROR "S09 self-mask patch no longer matches; review before building")
endif()
string(REPLACE "${old_mask}" "if(s13::selfPoint(v.x(),v.y(),v.z(),self_box_))continue;" terrain_source "${terrain_source}")
string(REPLACE "J cfg_;" "J cfg_;std::array<double,4> self_box_;" terrain_source "${terrain_source}")
string(REPLACE "f>>cfg_;" "f>>cfg_;self_box_=cfg_.at(\"self_mask\").get<std::array<double,4>>();" terrain_source "${terrain_source}")
string(REPLACE "std::array<int,2>cost" "std::array<int,2>confirm{{0,0}};std::array<int,2>cost" terrain_source "${terrain_source}")
string(REPLACE "Cell&c=cells_[i];c.t=t;" "Cell&c=cells_[i];const double observation_gap=t-c.t;c.t=t;" terrain_source "${terrain_source}")
string(REPLACE "if(s09::clear(c.cost[k]" "c.confirm[k]=s13::confirmLethal(c.confirm[k],cost==100,observation_gap);if(s09::clear(c.cost[k]" terrain_source "${terrain_source}")
string(REPLACE "SharedPtr,2>grids_;" "SharedPtr,2>grids_,confirmed_grids_;" terrain_source "${terrain_source}")
string(REPLACE "grids_[k++]=create_publisher" "confirmed_grids_[k]=create_publisher<nav_msgs::msg::OccupancyGrid>(\"/s13/terrain/confirmed/\"+name,2);grids_[k++]=create_publisher" terrain_source "${terrain_source}")
string(REPLACE "grids_[k]->publish(grid);" "grids_[k]->publish(grid);for(size_t i=0;i<cells_.size();++i)grid.data[i]=(t-cells_[i].t<.15&&cells_[i].confirm[k]>=3)?100:-1;confirmed_grids_[k]->publish(grid);" terrain_source "${terrain_source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/terrain_s13.cpp" "#include \"${CMAKE_CURRENT_SOURCE_DIR}/src/self_mask.hpp\"\n#include \"${CMAKE_CURRENT_SOURCE_DIR}/src/observation_confirmation.hpp\"\n${terrain_source}")
add_executable(terrain_s13 "${CMAKE_CURRENT_BINARY_DIR}/terrain_s13.cpp")
target_include_directories(terrain_s13 PRIVATE ../terrain_perception/include ${PCL_INCLUDE_DIRS})
ament_target_dependencies(terrain_s13 rclcpp sensor_msgs nav_msgs std_msgs pcl_conversions)
target_link_libraries(terrain_s13 ${PCL_LIBRARIES} Eigen3::Eigen nlohmann_json::nlohmann_json)
install(TARGETS command_gate clock_relay terrain_s13 RUNTIME DESTINATION lib/${PROJECT_NAME})
install(TARGETS measured_controller s13_drive LIBRARY DESTINATION lib)
ament_package()
