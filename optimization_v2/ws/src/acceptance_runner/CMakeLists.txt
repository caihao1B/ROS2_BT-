cmake_minimum_required(VERSION 3.16)
project(acceptance_runner)
find_package(ament_cmake REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(acceptance_monitor src/monitor.cpp)
target_compile_features(acceptance_monitor PRIVATE cxx_std_20)
target_compile_options(acceptance_monitor PRIVATE -Wall -Wextra -Werror)
target_include_directories(acceptance_monitor PRIVATE include)
target_link_libraries(acceptance_monitor nlohmann_json::nlohmann_json)
install(TARGETS acceptance_monitor DESTINATION lib/${PROJECT_NAME})
add_executable(gnss_supervisor17 src/supervisor17.cpp)
target_compile_features(gnss_supervisor17 PRIVATE cxx_std_17)
target_include_directories(gnss_supervisor17 PRIVATE include)
ament_target_dependencies(gnss_supervisor17 rclcpp geometry_msgs nav_msgs std_msgs std_srvs tf2_ros)
target_link_libraries(gnss_supervisor17 Eigen3::Eigen nlohmann_json::nlohmann_json)
install(TARGETS gnss_supervisor17 DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scripts tests include DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
ament_package()
