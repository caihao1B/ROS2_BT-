cmake_minimum_required(VERSION 3.16)
project(scenario_world)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
# Jazzy OR-Tools exports a newer protobuf config. Gazebo's Debian ABI is 3.21.
# Resolve the system MODULE before any Gazebo dependency creates imported targets.
set(Protobuf_PROTOC_EXECUTABLE /usr/bin/protoc CACHE FILEPATH "Gazebo system protoc" FORCE)
find_package(Protobuf MODULE REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz_transport_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
add_library(s02_motion SHARED src/ScriptedMotion.cc)
target_link_libraries(s02_motion gz-sim8::gz-sim8 gz-plugin2::gz-plugin2 gz-transport13::gz-transport13)
install(TARGETS s02_motion DESTINATION lib)
ament_package()
