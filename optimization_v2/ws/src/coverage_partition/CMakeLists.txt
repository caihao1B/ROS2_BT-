cmake_minimum_required(VERSION 3.16)
project(coverage_partition)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_library(METIS_LIBRARY NAMES metis REQUIRED)
add_executable(partition_metis src/partition_metis.cpp)
target_link_libraries(partition_metis ${METIS_LIBRARY})
install(TARGETS partition_metis DESTINATION lib/${PROJECT_NAME})
ament_package()
