cmake_minimum_required(VERSION 3.16)
project(localization_guard)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(diagnostic_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(robot_localization REQUIRED)
find_package(Eigen3 REQUIRED)
add_executable(health_guard src/health_guard.cpp)
ament_target_dependencies(health_guard rclcpp sensor_msgs nav_msgs std_msgs diagnostic_msgs)
target_link_libraries(health_guard nlohmann_json::nlohmann_json)
install(TARGETS health_guard DESTINATION lib/${PROJECT_NAME})
add_executable(ekf_probe ../../../lessons/S05/lesson.cpp)
ament_target_dependencies(ekf_probe robot_localization rclcpp)
find_library(RL_ENGINE rl_lib PATHS /opt/ros/jazzy/lib REQUIRED NO_DEFAULT_PATH)
target_link_libraries(ekf_probe Eigen3::Eigen ${RL_ENGINE})
install(TARGETS ekf_probe DESTINATION lib/${PROJECT_NAME})
ament_package()
