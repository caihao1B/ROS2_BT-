cmake_minimum_required(VERSION 3.16)
project(task_runtime)
find_package(ament_cmake REQUIRED)
add_library(r01_shutdown_compat SHARED src/shutdown_compat.cc)
target_compile_features(r01_shutdown_compat PRIVATE cxx_std_17)
target_link_libraries(r01_shutdown_compat PRIVATE -Wl,--no-as-needed boost_thread dl pthread)
target_link_options(r01_shutdown_compat PRIVATE -Wl,-z,nodelete)
install(TARGETS r01_shutdown_compat LIBRARY DESTINATION lib)
install(DIRECTORY scripts tests config DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
ament_package()
