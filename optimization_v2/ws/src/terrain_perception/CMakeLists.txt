cmake_minimum_required(VERSION 3.16)
project(terrain_perception)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2)
find_package(ament_cmake REQUIRED)
# Resolve Gazebo's system Protobuf ABI before Nav2/OR-Tools exports its config.
set(Protobuf_PROTOC_EXECUTABLE /usr/bin/protoc CACHE FILEPATH "Gazebo system protoc" FORCE)
find_package(Protobuf MODULE REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(PCL REQUIRED COMPONENTS common filters segmentation search)
find_package(Eigen3 REQUIRED)
find_package(nlohmann_json REQUIRED)
remove_definitions(EIGEN_MPL2_ONLY)
add_compile_definitions(EIGEN_MPL2_ONLY)
add_executable(terrain_node src/terrain_node.cpp)
target_include_directories(terrain_node PRIVATE include ${PCL_INCLUDE_DIRS})
ament_target_dependencies(terrain_node rclcpp sensor_msgs nav_msgs std_msgs visualization_msgs pcl_conversions)
target_link_libraries(terrain_node ${PCL_LIBRARIES} Eigen3::Eigen nlohmann_json::nlohmann_json)
add_library(observed_risk_layer SHARED src/risk_layer.cpp)
ament_target_dependencies(observed_risk_layer rclcpp nav_msgs nav2_costmap_2d pluginlib)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugins.xml)
install(TARGETS terrain_node observed_risk_layer RUNTIME DESTINATION lib/${PROJECT_NAME} LIBRARY DESTINATION lib)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(s09_actor SHARED src/Actor.cc)
target_link_libraries(s09_actor gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS s09_actor LIBRARY DESTINATION lib)
ament_package()
