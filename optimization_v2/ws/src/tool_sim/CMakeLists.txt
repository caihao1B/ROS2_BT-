cmake_minimum_required(VERSION 3.16)
project(tool_sim)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2)
find_package(ament_cmake REQUIRED)
set(Protobuf_PROTOC_EXECUTABLE /usr/bin/protoc CACHE FILEPATH "Gazebo system ABI" FORCE)
find_package(Protobuf MODULE REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-transport13 REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(mission_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rosgraph_msgs REQUIRED)
add_library(s10_tool SHARED src/ToolActuator.cc)
target_link_libraries(s10_tool gz-sim8::gz-sim8 gz-plugin2::gz-plugin2 gz-transport13::gz-transport13 nlohmann_json::nlohmann_json)
add_executable(tool_action_server src/action_server.cpp)
target_include_directories(tool_action_server PRIVATE include)
ament_target_dependencies(tool_action_server rclcpp rclcpp_action mission_interfaces std_msgs rosgraph_msgs)
target_link_libraries(tool_action_server nlohmann_json::nlohmann_json)
install(TARGETS s10_tool LIBRARY DESTINATION lib)
install(TARGETS tool_action_server RUNTIME DESTINATION lib/${PROJECT_NAME})
ament_package()
